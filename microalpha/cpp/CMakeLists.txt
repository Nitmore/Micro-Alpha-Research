cmake_minimum_required(VERSION 3.16)
project(microalpha_book LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W4 /O2 /EHsc /permissive-)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  add_compile_options(-Wall -Wextra -O2)
endif()
add_library(mabook STATIC src/book.cpp)
target_include_directories(mabook PUBLIC include)
add_executable(book_builder src/book_builder.cpp)
target_link_libraries(book_builder PRIVATE mabook)
add_executable(test_book tests/test_book.cpp)
target_link_libraries(test_book PRIVATE mabook)
enable_testing()
add_test(NAME book_unit_tests COMMAND test_book)
